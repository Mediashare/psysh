REPL scope.
      - Always finally-stop profiler and always emit trace path (subprocess).

    function_key: string   # fully-qualified name or "Class::method"

        - Returns normalized_profile_schema

          user: exclude namespaces Psy\, PhpParser\, Symfony\Console\, Symfony\VarDumper
          php: user + internal PHP functions
          all: no namespace filter
        - Threshold: μs; negative => 0

    - name: Rendering & options

        --filter: user|php|all
        --threshold: int μs; applied after level filter

        --out: write normalized JSON (unfiltered) to path
        --debug: print engine, script (subprocess), command, trace path, parsed totals
      outputs:
        - Table sorted by time desc; adaptive units for time/memory

    - name: Error handling & cleanup
      rules:
        - If no xdebug/xhprof: RuntimeException with actionable hint

        - finally: unlink temp script and trace (unless --debug)
